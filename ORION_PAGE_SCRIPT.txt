    function parsePricingCSV(text) {
        const lines = text.split(/\r?\n/);
        const matrices = {};
        let currentKey = null;
        let currentWidths = null;
        for (let i = 0; i < lines.length; i++) {
            const raw = lines[i] == null ? '' : lines[i];
            const trimmed = raw.trim();
            if (trimmed === '') continue;
            if (trimmed.startsWith('#')) {
                const m = trimmed.match(/^#(GROUP|TARGET):(.+)$/i);
                if (m) {
                    currentKey = m[2].split(',')[0].trim();
                    currentWidths = null;
                    matrices[currentKey] = {};
                }
                continue;
            }
            const fields = parseCSVRow(raw);
            if (currentKey && currentWidths == null) {
                const firstIsEmpty = (fields[0] || '').trim() === '';
                if (firstIsEmpty) {
                    const numericValues = fields.slice(1).map(f => f.trim() === '' ? null : parseFloat(f)).filter(v => v !== null && !isNaN(v));
                    if (numericValues.length > 0) {
                        currentWidths = numericValues;
                        continue;
                    }
                }
            }
            if (currentKey && currentWidths) {
                const heightVal = parseFloat(fields[0]);
                if (isNaN(heightVal)) continue;
                const row = {};
                const missingWidths = [];
                for (let wIdx = 0; wIdx < currentWidths.length; wIdx++) {
                    const fieldIdx = wIdx + 1;
                    const f = (fields[fieldIdx] || '').trim();
                    if (f === '') {
                        missingWidths.push(currentWidths[wIdx]);
                        continue;
                    }
                    const price = parseFloat(f);
                    if (!isNaN(price)) {
                        row[currentWidths[wIdx]] = price;
                    } else {
                        missingWidths.push(currentWidths[wIdx]);
                    }
                }
                if (missingWidths.length > 0) {
                    missingWidths.forEach(w => {
                        console.error('[Orion] CRITICAL: Matrix "' + currentKey + '" is missing price for ' + w + 'x' + heightVal + '. Please fix the pricing CSV.');
                    });
                }
                if (Object.keys(row).length > 0) matrices[currentKey][heightVal] = row;
            }
        }
        return matrices;
    }

    function getStandardHeadrailDisplayString(prod) {
        if (!prod) return '';
        const standardHeadrailId = (prod.Standard_Headrail_Accessory_ID || '').toString().trim();
        if (!standardHeadrailId) return '';
        const acc = STATE.data.accessories.find(a => a.Accessory_ID === standardHeadrailId);
        if (!acc) {
            console.warn('[Orion] Standard headrail accessory "' + standardHeadrailId + '" not found in accessories CSV. _Orion_Headrail will be empty.');
            return '';
        }
        const sub = STATE.selections.accSubSelections[standardHeadrailId] || getDefaultSubOption(acc);
        const colorId = STATE.selections.accColorSelections[standardHeadrailId];
        const color = colorId ? getColorForAccessory(acc, colorId) : null;

        if (sub) refinements.push(sub);
        if (color && color.Color_Name) refinements.push(color.Color_Name);

    function buildOrionAccessoryProperties(prod) {
        const props = {};
        const displayFor = (accId) => {
            const acc = STATE.data.accessories.find(a => a.Accessory_ID === accId);
            if (!acc) return '';
            return buildAccessoryDisplayString(acc, accId);
        };
        const selectedHeadrailId = STATE.selections.accessorySelections['Headrail'];
        if (selectedHeadrailId) {
            props['_Orion_Headrail'] = displayFor(selectedHeadrailId);
        } else {
            props['_Orion_Headrail'] = getStandardHeadrailDisplayString(prod);
        }
        const categoryBuckets = [
            { cat: 'Remote', key: '_Orion_Remote' },
            { cat: 'Charger', key: '_Orion_Charger' },
            { cat: 'Smart Automation', key: '_Orion_Smart_Automation' }
        ];
        categoryBuckets.forEach(bucket => {
            const selectedId = STATE.selections.accessorySelections[bucket.cat];
            if (!selectedId) return;
            const acc = STATE.data.accessories.find(a => a.Accessory_ID === selectedId);
            if (!acc) return;
            const actualCategory = (acc.Accessory_Category || '').toString().trim();
            if (actualCategory !== bucket.cat) {
                console.warn('[Orion] Accessory ' + selectedId + ' (' + (acc.Accessory_Name || '') + ') is in category "' + actualCategory + '" but was selected under "' + bucket.cat + '". Leaving ' + bucket.key + ' blank; accessory stays in Hardware.');
                return;
            }
            props[bucket.key] = displayFor(selectedId);
        });
        return props;
    }

        };

            if(e) e.preventDefault();

                const missing = getMissingSelections();

                showToast('Sorry, this product is not available for online purchase yet.');

            const results = await Promise.allSettled(keys.map(k => fetch(csvUrl(CSV_URLS[k]), { cache: 'no-store' }).then(r => { if (!r.ok) throw new Error('HTTP ' + r.status + ' on ' + k + ' (GitHub CSV fetch failed)'); return r.text(); })));
            const errors = [];
            const data = {};
            keys.forEach((k, i) => { const r = results[i]; if (r.status === 'fulfilled') { data[k] = (k === 'pricing') ? parsePricingCSV(r.value) : parseCSV(r.value); } else { errors.push(r.reason.message); data[k] = (k === 'pricing') ? {} : []; } });
            if (errors.length > 0) { const errEl = el.error(); const formEl = el.form(); const loaderEl = el.loader(); if (loaderEl) loaderEl.style.display = 'none'; if (formEl) formEl.classList.add('hidden'); if (errEl) { errEl.classList.remove('hidden'); errEl.classList.add('flex'); const p = errEl.querySelector('p'); if (p) p.innerText = errors.join(' • '); } return; }
            const sanityFailures = runSanityGates(data);
            if (sanityFailures.length > 0) { const errEl = el.error(); const formEl = el.form(); const loaderEl = el.loader(); if (loaderEl) loaderEl.style.display = 'none'; if (formEl) formEl.classList.add('hidden'); if (errEl) { errEl.classList.remove('hidden'); errEl.classList.add('flex'); const p = errEl.querySelector('p'); if (p) p.innerText = 'Required data is missing or empty: ' + sanityFailures.join(' • '); } return; }
            STATE.data.productUi = data.productUi || [];
            STATE.data.fabricTypes = data.fabricTypes || [];
            STATE.data.productLifts = data.productLifts || [];
            STATE.data.productAccessories = data.productAccessories || [];
            STATE.data.productAccColors = data.productAccColors || [];
            STATE.data.lifts = data.lifts || [];
            STATE.data.accessories = data.accessories || [];
            STATE.data.accColors = data.accColors || [];
            STATE.data.colors = {};
            (data.colors || []).forEach(c => { const cid = c.Collection_ID; if (!cid) return; if (!STATE.data.colors[cid]) STATE.data.colors[cid] = []; STATE.data.colors[cid].push(c); });
            STATE.data.pricing = data.pricing || {};
            const ctxErr = resolveContext();
            if (ctxErr) { showContextError(ctxErr); return; }
            STATE.data.products = (data.products || []).filter(p => { return (p.Product_Type || '').trim() === CONTEXT.productType && (p.Vendor || '').trim() === CONTEXT.vendor; });
            const contextFailures = runContextSanityChecks();
            if (contextFailures.length > 0) { showContextError({ code: 'CONTEXT_VALIDATION', failures: contextFailures }); return; }
            STATE.data.products.forEach(p => { const ui = CONTEXT.uiRow; if (ui) { p.Lift_Family_Image_URL = (ui.Lift_Family_Image_URL || '').toString().trim(); p.Accessory_Family_Image_URL = (ui.Accessory_Family_Image_URL || '').toString().trim(); p.Mount_Comparison_Image_URL = (ui.Mount_Comparison_Image_URL || '').toString().trim(); p.Mount_Inside_Image_URL = (ui.Mount_Inside_Image_URL || '').toString().trim(); p.Mount_Outside_Image_URL = (ui.Mount_Outside_Image_URL || '').toString().trim(); } else { p.Lift_Family_Image_URL = ''; p.Accessory_Family_Image_URL = ''; p.Mount_Comparison_Image_URL = ''; p.Mount_Comparison_Image_URL = ''; p.Mount_Outside_Image_URL = ''; } });
            STATE.data.lifts.forEach(l => { l.ProductLiftImages = {}; });
            STATE.data.productLifts.forEach(row => { if (row.Product_Type && row.Lift_ID) { const url = (row.Image_URL || '').toString().trim(); const lift = STATE.data.lifts.find(l => l.Lift_ID === row.Lift_ID); if (lift) lift.ProductLiftImages[row.Product_Type] = url; } });
            STATE.data.accessories.forEach(a => { a.ProductAccessoryImages = {}; });
            STATE.data.productAccessories.forEach(row => { if (row.Product_Type && row.Accessory_ID) { const url = (row.Image_URL || '').toString().trim(); const acc = STATE.data.accessories.find(a => a.Accessory_ID === row.Accessory_ID); if (acc) acc.ProductAccessoryImages[row.Product_Type] = url; } });
            STATE.data.productAccColorImages = {};
            STATE.data.productAccColors.forEach(row => { if (row.Product_Type && row.Accessory_ID && row.Color_ID) { const key = row.Product_Type + '|' + row.Accessory_ID + '|' + row.Color_ID; STATE.data.productAccColorImages[key] = (row.Image_URL || '').toString().trim(); } });
            STATE.data.fabricTypeByKey = {};
            STATE.data.fabricTypes.forEach(ft => { if (ft.Product_Type && ft.Fabric_Type) { const key = ft.Product_Type + '|' + ft.Fabric_Type; STATE.data.fabricTypeByKey[key] = ft; } });
            if (!runPricingAssertions()) return;
            STATE.selections.fabricType = CONTEXT.fabricType;
            applyContextText();
            console.log('[Orion] CONTEXT:', { productSlug: CONTEXT.productSlug, productType: CONTEXT.productType, vendor: CONTEXT.vendor, fabricSlug: CONTEXT.fabricSlug, fabricType: CONTEXT.fabricType, fabricTypeRowCount: CONTEXT.fabricTypeRows.length });
            updateLoader('Building your shade...');
            rebindLifestyleHeroes();
            setSolarHero(STATE.selections.fabricType);
            populateSizeDropdowns();
            determineVisibleCategories();
            renderUI();
            renderMotorizedAccessories();
            renderAestheticCategories();
            rebuildLiftLayer();
            updatePricing();
            applySectionGates();
            setupSwatchAutoHide();
            setupSwatchRailOverflow();
            setupPaletteStrip();
            setupSwatchDetailClose();
            setupSectionProgress();
            setupReviewToggle();
            setupCheckout();
            setupViewportStateMachine();
            syncMountModalImages();
            try { const origOpen = window.openMountModal; window.openMountModal = function() { if (typeof origOpen === 'function') { try { origOpen(); } catch (e) { console.warn('openMountModal failed', e); } } syncMountModalImages(); }; } catch (e) { console.warn('openMountModal wrap failed', e); }
            const loaderEl = el.loader();
            const formEl = el.form();
            if (loaderEl) loaderEl.style.display = 'none';
            if (formEl) formEl.classList.remove('hidden');
        } catch (e) {
            console.error('init failed', e);
            const loaderEl = el.loader();
            const formEl = el.form();
            const errEl = el.error();
            if (loaderEl) loaderEl.style.display = 'none';
            if (formEl) formEl.classList.add('hidden');
            if (errEl) { errEl.classList.remove('hidden'); errEl.classList.add('flex'); const p = errEl.querySelector('p'); if (p) p.innerText = 'Unexpected error: ' + (e.message || 'see console'); }
        }
    }

    const setupListeners = () => {
        const configForm = el.form();
        if (!configForm) return;
        const handleSizeChange = () => {
            STATE.hasUserInteracted = true;
            const widthEl = document.getElementById('width-whole');
            const heightEl = document.getElementById('height-whole');
            const widthFracEl = configForm.querySelector('select[name="width_fraction"]');
            const heightFracEl = configForm.querySelector('select[name="height_fraction"]');
            let wholeW = STATE.selections.width;
            let wholeH = STATE.selections.height;
            if (widthEl) { const parsedW = parseInt(widthEl.value, 10); if (!isNaN(parsedW) && parsedW > 0) wholeW = parsedW; }
            if (heightEl) { const parsedH = parseInt(heightEl.value, 10); if (!isNaN(parsedH) && parsedH > 0) wholeH = parsedH; }
            const wEighths = parseFractionValue(widthFracEl ? widthFracEl.value : '0');
            const hEighths = parseFractionValue(heightFracEl ? heightFracEl.value : '0');
            STATE.selections.width = wholeW;
            STATE.selections.height = wholeH;
            STATE.selections.widthEighths = wEighths;
            STATE.selections.heightEighths = hEighths;
            STATE.selections.actualWidth = wholeW + (wEighths / EIGHTHS);
            STATE.selections.actualHeight = wholeH + (hEighths / EIGHTHS);
            determineVisibleCategories();
            revalidateSelections();
            renderUI();
            renderMotorizedAccessories();
            renderAestheticCategories();
            rebuildLiftLayer();
            updatePricing();
            applySectionGates();
            setupPaletteStrip();
        };
        configForm.addEventListener('change', (e) => {
            const t = e.target;
            if (t && (t.id === 'width-whole' || t.id === 'height-whole')) handleSizeChange();
            if (t && t.name === 'width_fraction') handleSizeChange();
            if (t && t.name === 'height_fraction') handleSizeChange();
        });
        configForm.onchange = (e) => {
            const n = e.target.name;
            if (n === 'fabric') {
                STATE.hasUserInteracted = false;
                const newFabricType = e.target.value;
                STATE.selections.fabricType = newFabricType;
                const fabricTypeRow = CONTEXT.fabricTypeRows.find(ft => (ft.Fabric_Type || '').trim() === newFabricType);
                if (fabricTypeRow) { CONTEXT.fabricType = newFabricType; CONTEXT.fabricSlug = (fabricTypeRow.Fabric_Type_Slug || '').trim(); }
                STATE.showAllCollections = false;
                STATE.selections.collectionId = '';
                STATE.selections.colorName = '';
                hideSwatchDetail();
                hideCollectionImage();
                setSolarHero(STATE.selections.fabricType);
                populateSizeDropdowns();
                revalidateSelections();
                determineVisibleCategories();
                renderUI();
                renderMotorizedAccessories();
                renderAestheticCategories();
                rebuildLiftLayer();
                updatePricing();
                applySectionGates();
                setupPaletteStrip();
            } else if (n === 'color') {
                STATE.hasUserInteracted = true;
                const val = e.target.value || '';
                const [cid, color] = val.split('|');
                if (cid) {
                    if (STAGED_SELECTION_ENABLED) { selectCollectionOnly(cid); return; }
                    STATE.selections.collectionId = cid;
                    STATE.selections.colorName = color;
                    const colors = STATE.data.colors[cid] || [];
                    const cObj = colors.find(c => c.Color_Name === color);
                    if (cObj && cObj.Image_URL) showSwatchDetail(cObj.Image_URL);
                    populateSizeDropdowns();
                    revalidateSelections();
                    determineVisibleCategories();
                    renderUI();
                    renderMotorizedAccessories();
                    renderAestheticCategories();
                    rebuildLiftLayer();
                    updatePricing();
                    applySectionGates();
                    setupPaletteStrip();
                }
            } else if (n === 'mount') {
                STATE.hasUserInteracted = true;
                STATE.selections.mount = e.target.value;
                setMountView(STATE.selections.mount);
                updatePricing();
                applySectionGates();
            } else if (n === 'lift') {
                STATE.hasUserInteracted = true;
                STATE.selections.liftId = e.target.value;
                setLiftView(STATE.selections.liftId);
                determineVisibleCategories();
                (STATE.motorizedCategories || []).forEach(cat => { STATE.selections.accessorySelections[cat] = null; });
                revalidateSelections();
                renderUI();
                renderMotorizedAccessories();
                renderAestheticCategories();
                updatePricing();
                applySectionGates();
            } else if (n === 'measurement' || n === 'installation') {
                STATE.hasUserInteracted = true;
                STATE.selections[n] = e.target.value;
                updatePricing();
                applySectionGates();
            }
        };
        configForm.oninput = (e) => {
            if (e.target.id === 'width-whole' || e.target.id === 'height-whole') handleSizeChange();
            else if (e.target.name && e.target.name.startsWith('width_fraction')) { STATE.hasUserInteracted = true; updatePricing(); applySectionGates(); }
            else if (e.target.name && e.target.name.startsWith('height_fraction')) { STATE.hasUserInteracted = true; updatePricing(); applySectionGates(); }
        };
    };

    window.__orionInit = init;
    window.__selectCollectionOnly = selectCollectionOnly;
    window.__selectColorFromGrid = selectColorFromGrid;
    window.__showCollectionImage = showCollectionImage;

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => { init(); setupListeners(); });
    } else {
        init();
        setupListeners();
    }
})();</script>
